QUICK COMMAND REFERENCE

Installation:
  mkdir -p ~/.pi/agent/extensions/subagent
  ln -sf $(pwd)/examples/extensions/subagent/* ~/.pi/agent/extensions/subagent/

  mkdir -p ~/.pi/agent/agents
  for f in examples/extensions/subagent/agents/*.md; do
    ln -sf $(pwd)/$f ~/.pi/agent/agents/$(basename "$f")
  done

Interactive:
  pi
  > /implement add Redis caching
  > Use scout to find auth code
  > Run 3 scouts: auth, models, providers

Single Agent:
  {
    "agent": "scout",
    "task": "Find all error handlers"
  }

Parallel:
  {
    "tasks": [
      { "agent": "scout", "task": "Find auth" },
      { "agent": "scout", "task": "Find models" }
    ]
  }

Chain:
  {
    "chain": [
      { "agent": "scout", "task": "Find code" },
      { "agent": "planner", "task": "Plan: {previous}" },
      { "agent": "worker", "task": "Implement: {previous}" }
    ]
  }
